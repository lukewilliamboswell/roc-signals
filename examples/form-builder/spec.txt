# Form Builder — native semantic spec.

# The app has two levels of state: the designer's schema and the answers typed
# into the generated preview. Every assertion below about a preview label, a
# preview error, or the submittable flag is an assertion about derived state:
# nothing in the app writes those values from an event handler.

# -------------------------------------------- answering the generated form
# One answer at a time: the submittable fan-in only flips when BOTH inputs are
# satisfied.

expect_text text:"Preview status: 2 fields, 1 problem" "Preview status: 2 fields, 1 problem"

expect_text text:"Form is not submittable" "Form is not submittable"

# ------------------------------------- the key cross-level assertion
# Tighten a rule in the DESIGNER and watch previously-valid PREVIEW input turn
# invalid, with the submittable signal flipping in the same propagation.
fill label:"Minimum f1" "5"
expect_text text:"Full name: Must be at least 5 characters" "Full name: Must be at least 5 characters"
expect_attr label:"Answer f1" aria-invalid ""
expect_value label:"Answer f1" "Ada"
expect_text text:"Preview status: 2 fields, 1 problem" "Preview status: 2 fields, 1 problem"
expect_text text:"Form is not submittable" "Form is not submittable"
expect_disabled role:button name:"Submit form" true

# Relaxing the same rule restores validity without touching the answer.
fill label:"Minimum f1" "2"
expect_text text:"Full name: Valid" "Full name: Valid"
expect_text text:"Form is submittable" "Form is submittable"
expect_disabled role:button name:"Submit form" false
