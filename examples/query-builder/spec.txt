# Query Builder: a recursive AND/OR filter tree.
#
# Documented semantics locked below:
#  - a condition with an empty value imposes no constraint and matches every row
#    (it renders as ANY);
#  - an empty group imposes no constraint and matches every row in both AND and
#    OR mode (it renders as ALL), so a negated empty group matches nothing;
#  - > and < only apply to the numeric Level field; on a text field they match
#    nothing.

# ---------------------------------------------------------------- initial state
expect_visible role:heading name:"Query Builder"
expect_visible role:region name:"Dataset"
expect_visible role:region name:"Generated query"
expect_visible role:region name:"Matches"
expect_visible role:region name:"Filter tree"
expect_visible role:region name:"Group n1"
expect_visible role:region name:"Condition n2"
expect_checked label:"Include archived rows" false
expect_text test_id:"dataset-summary" "Dataset rows: 5"
expect_text test_id:"query-text" "(dept = 'Platform')"
expect_text test_id:"shape-stats" "Conditions 1 / Groups 1 / Depth 1"
expect_text test_id:"match-summary" "Matching rows: 2 of 5"
expect_text test_id:"summary-n1" "(dept = 'Platform')"
expect_text test_id:"summary-n2" "dept = 'Platform'"
expect_value label:"Mode n1" "AND"
expect_checked label:"Negate n1" false
expect_value label:"Field n2" "dept"
expect_value label:"Operator n2" "eq"
expect_value label:"Value n2" "Platform"
expect_visible test_id:"match-Ada"
expect_visible test_id:"match-Bo"
expect_absent test_id:"match-Cy"
expect_absent test_id:"match-Dee"
expect_absent test_id:"match-Eli"
# The root group is not deletable.
expect_absent role:button name:"Delete n1"
expect_text text:"Root group cannot be deleted" "Root group cannot be deleted"

# ------------------------------------------------- edge case: empty leaf value
fill label:"Value n2" ""
expect_text test_id:"summary-n2" "ANY"
expect_text test_id:"query-text" "(ANY)"
expect_text test_id:"match-summary" "Matching rows: 5 of 5"
expect_visible test_id:"match-Cy"
fill label:"Value n2" "Platform"
expect_text test_id:"match-summary" "Matching rows: 2 of 5"

# ------------------------------------------------------------- every operator
# eq is the initial operator and is asserted above.
select_option label:"Operator n2" "ne"
expect_text test_id:"query-text" "(dept != 'Platform')"
expect_text test_id:"match-summary" "Matching rows: 3 of 5"
expect_absent test_id:"match-Ada"
expect_visible test_id:"match-Cy"
expect_visible test_id:"match-Dee"
expect_visible test_id:"match-Eli"

select_option label:"Operator n2" "contains"
fill label:"Value n2" "esi"
expect_text test_id:"query-text" "(dept ~ 'esi')"
expect_text test_id:"match-summary" "Matching rows: 2 of 5"
expect_visible test_id:"match-Cy"
expect_visible test_id:"match-Eli"

# > and < on a text field match nothing.
select_option label:"Field n2" "name"
select_option label:"Operator n2" "gt"
fill label:"Value n2" "3"
expect_text test_id:"query-text" "(name > '3')"
expect_text test_id:"match-summary" "Matching rows: 0 of 5"
expect_absent test_id:"match-Ada"

select_option label:"Field n2" "level"
expect_text test_id:"query-text" "(level > '3')"
expect_text test_id:"match-summary" "Matching rows: 2 of 5"
expect_visible test_id:"match-Ada"
expect_visible test_id:"match-Cy"

select_option label:"Operator n2" "lt"
expect_text test_id:"query-text" "(level < '3')"
expect_text test_id:"match-summary" "Matching rows: 2 of 5"
expect_visible test_id:"match-Bo"
expect_visible test_id:"match-Dee"

# A non-numeric value on a numeric comparison matches nothing.
fill label:"Value n2" "three"
expect_text test_id:"query-text" "(level < 'three')"
expect_text test_id:"match-summary" "Matching rows: 0 of 5"

# Back to the starting condition.
select_option label:"Field n2" "dept"
select_option label:"Operator n2" "eq"
fill label:"Value n2" "Platform"
expect_text test_id:"query-text" "(dept = 'Platform')"
expect_text test_id:"match-summary" "Matching rows: 2 of 5"

# ------------------------------------------- second independent source fan-in
mark_metrics
check label:"Include archived rows"

expect_text test_id:"dataset-summary" "Dataset rows: 5"

expect_absent test_id:"match-Fay"

# ---------------------------------------------- nesting: a group inside a group
mark_metrics

expect_metric_delta rows_reused 1

# A negated empty group matches nothing.
check label:"Negate n3"

expect_text test_id:"match-summary" "Matching rows: 0 of 5"
uncheck label:"Negate n3"
expect_text test_id:"match-summary" "Matching rows: 2 of 5"

expect_text test_id:"shape-stats" "Conditions 2 / Groups 2 / Depth 2"
expect_text test_id:"match-summary" "Matching rows: 2 of 5"

# ------------------------------------------------------- AND / OR on a group

expect_visible test_id:"match-Bo"

expect_value label:"Value n5" "o"

# ------------------------------------------------------ three levels of nesting

expect_text test_id:"match-summary" "Matching rows: 2 of 5"

# Switching the middle group to AND changes the match count.

expect_text test_id:"match-summary" "Matching rows: 2 of 5"

# --------------- editing a leaf three levels deep does not recreate its siblings
