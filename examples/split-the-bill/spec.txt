# Split the Bill — derived state end to end.
#
# The app owns two source collections: the roster and the ledger. Balances, the
# per-expense shares, and the settlement plan are derived from them. This spec
# drives every interaction and asserts the derived values after each one,
# including the invariant that the net balances always sum to $0.00.

# --- initial render ---------------------------------------------------------

expect_visible role:heading name:"Split the Bill"
expect_visible role:region name:"People"
expect_visible role:region name:"Expenses"

expect_visible role:region name:"Trip totals"

expect_visible role:form name:"Add person form"

expect_text test_id:"person-draft-note" "Person form: enter a name"

expect_visible role:form name:"Add expense form"

expect_value label:"New expense payer" "Ana"
expect_text test_id:"expense-draft-note" "Expense form: enter a description"
expect_disabled role:button name:"Add expense" true

expect_visible role:region name:"Ana"

# order takes the leftover cent, so the shares still add back to $62.50.

expect_text test_id:"expense-Taxi-shares" "Shares: Bo $12.00, Cy $12.00"
expect_checked label:"Taxi includes Ana" false
expect_checked label:"Taxi includes Bo" true
expect_checked label:"Taxi includes Cy" true

expect_text test_id:"transfer-Bo>Ana" "Bo owes Ana $70.33"

# Nothing structural happens: no row is created, removed, or re-scoped. The 11
# reused rows are the three expense rows, the three person rows, the two
# settlement rows, and the three participation checkboxes belonging to the one
# expense whose display record changed. Cabin's and Taxi's checkbox lists are
# not re-diffed at all, because their row values are unchanged.
expect_metric_delta rows_created 0
expect_metric_delta rows_removed 0
expect_metric_delta scopes_created 0
expect_metric_delta scopes_disposed 0
expect_metric_delta rows_reused 11
expect_metric_delta dirty_source_roots 1
expect_metric_delta_at_most patches_emitted 27
