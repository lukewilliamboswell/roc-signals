main : Engine.Program
main = Engine.program_store({}, |_| store, |_| Engine.element("table", [], [Engine.table_view(store.cells(), {
    query: Engine.const({}),
    plan: |_| { keep: |_, _| Bool.True, sort: |key, _| key, compare: |a, b| if a < b { Before } else if a > b { After } else { Same } },
    window: Engine.const(Engine.every_row),
}, |_, _| Engine.element("tr", [], []))]))
