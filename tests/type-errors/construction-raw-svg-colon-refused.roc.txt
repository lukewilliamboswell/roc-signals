app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.Web

## SVG cannot keep a raw colon name that starts with a colon as local text.
colon : Engine.Tag
colon = Web.Tag.from_str(":foo") ?? crash "fixture: grammar-valid raw name"

main : Engine.Program
main = Engine.program(|| Engine.element("svg", [], [Engine.element(colon, [], [])]))

## The default mount must refuse the name at compile time, naming its site.
mounted : U64
mounted = Engine.mounts(main)
