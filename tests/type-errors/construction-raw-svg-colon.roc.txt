import pf.Engine
import pf.Web

## A colon in a raw SVG name is local text, never a prefix, so these mount.
raw : Str -> Engine.Elem(Engine.Root)
raw = |text| Engine.element(Web.Tag.from_str(text) ?? crash "fixture: grammar-valid raw name", [], [])

main : Engine.Program
main = Engine.program(|| Engine.element("svg", [], [raw("xlink:x"), raw("xml:foo"), raw("xmlns:foo")]))

## These construction fixtures have no shared store.
Store : {}

## The default mount constructs every raw name at compile time.
mounted : U64
mounted = Engine.mounts(main)
