app [main] { pf: platform "<PLATFORM>" }

import pf.Engine

## A row's text hole directly in a table is refused when its template is
## compiled, even while every row holds only whitespace.
main : Engine.Program
main = Engine.program(|| Engine.each(Engine.const([" "]), |_| "one", |row| Engine.element("table", [], [Engine.text_s(row.item())])))

## Check the default first page while compiling.
mounted : U64
mounted = Engine.mounts(main)
