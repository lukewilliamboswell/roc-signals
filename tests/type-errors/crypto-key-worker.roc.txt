app [main] { pf: platform "<PLATFORM>" }
import pf.Engine
import pf.WebCrypto
import pf.Worker

task : Worker.Task(WebCrypto.Key, U64, Worker.Never)
task = Worker.task({ name: "key", run!: |_key| Ok(1) })

main : Engine.Program
main = Engine.program(|| Engine.text("key")).with_workers([task.registration()])
