app [main] { pf: platform "<PLATFORM>" }
import pf.Engine
import pf.Html
Ctx : Engine.Root
view : Engine.Elem(Ctx)
view = {
	assemble : Iter(Engine.Signal(Ctx, Str)) -> Html.Markup(Ctx)
	assemble = Html.Markup.from_interpolation(["<p>", "</p>"]) ?? crash "valid fixed source"
	markup = assemble(List.iter([Engine.const("a"), Engine.const("b")]))
	markup.to_elem()
}
main : Engine.Program
main = Engine.program(|| view)
Store : {}
mounted : U64
mounted = Engine.mounts(main)
