app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.Html

## A literal table row placed in a div has no table context.
view : Engine.Elem(Engine.Root)
view = {
	markup : Html.Markup(Engine.Root)
	markup = "<tr><td>x</td></tr>"
	Engine.element("div", [], [markup.to_elem()])
}

main : Engine.Program
main = Engine.program(|| view)

## The default mount must refuse the placement at compile time.
mounted : U64
mounted = Engine.mounts(main)
