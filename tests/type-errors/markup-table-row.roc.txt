app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.Html

## A table row is a literal root of its own, drawn as each row of a table body.
## Its context is concrete so roc-lang/roc#13015 cannot hide a rejection.
row : Engine.RowRef(Engine.Root, Str) -> Engine.Elem(Engine.Row(Engine.Root, Str))
row = |item| {
	markup : Html.Markup(Engine.Row(Engine.Root, Str))
	markup = "<tr><td>${item.key()}</td></tr>"
	markup.to_elem()
}

main : Engine.Program
main = Engine.program(|| Engine.element("table", [], [Engine.element("tbody", [], [Engine.each(Engine.const(["a", "b"]), |name| name, row)])]))

## These construction fixtures have no shared store.
Store : {}

## The default mount constructs the row template and draws both rows.
mounted : U64
mounted = Engine.mounts(main)
