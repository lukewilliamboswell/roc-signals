app [main] { pf: platform "<PLATFORM>" }

import pf.Engine

## Rows written in the program: the compiler evaluates this constant, so a
## repeated key is a compile error naming it.
rows : Engine.Table(U64, Str)
rows = match Engine.Table.from_list([{ key: 1.U64, row: "one" }, { key: 1, row: "two" }]) {
	Ok(table) => table
	Err(DuplicateKey(key)) => crash "rows lists the key ${key.to_str()} twice"
}

main : Engine.Program
main = Engine.program(|| Engine.text(rows.len().to_str()))

## This fixture has no shared store.
Store : {}
