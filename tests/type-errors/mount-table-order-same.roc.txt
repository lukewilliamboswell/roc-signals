app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.Html

Tag := [App]
store = Engine.store(App, Engine.table([{ key: 1.U64, row: 1.U64 }, { key: 2, row: 2 }]))

main : Engine.Program
main = Engine.program_store({}, |_| store, |_| Engine.table_view(store.cells(), {
 query: Engine.const({}),
 plan: |_| { keep: |_, _| Bool.True, sort: |_, row| row, compare: |_, _| Same },
 window: Engine.const(Engine.every_row),
}, |_, row| Html.li([], [Engine.text_s(row.signal().map(U64.to_str))])))

## The table's generated store storage type.
Store : Engine.Stored(Tag, Engine.Table(U64, U64))

## Check the default first page.
mounted : U64
mounted = Engine.mounts(main)
