app [main] { pf: platform "<PLATFORM>" }

import pf.Engine

Tag := [App]
store = Engine.store(App, Engine.table([{key:1.U64,row:1.U64}]))

main : Engine.Program
main = Engine.program_store({}, |_| store, |_| Engine.table_view(store.cells(), {
 query: Engine.const({}),
 plan: |_| { keep: |_, _| Bool.True, sort: |key, _| key, compare: |a, b| if a < b { Before } else if a > b { After } else { Same } },
 window: Engine.const(Engine.every_row),
}, |_, _| Engine.text("not an element")))

## The table's generated storage type.
Store : Engine.Stored(Tag, Engine.Table(U64,U64))

## Check the default first page.
mounted : U64
mounted = Engine.mounts(main)
