app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.Port

## A handwritten port codec is generic over its format: it passes the
## encoder's output on through the hooks it names, never building one.
Words := List(Str).{
 parser_for : _
 encoder_for : format -> (Words, enc -> Try(enc, []))
  where [
   format.encode_list : enc, U64, (items, (items, (enc -> Try(enc, [])) -> Try(items, [])) -> Try(items, [])) -> Try(enc, []),
   format.encode_str : Str, enc -> Try(enc, []),
  ]
 encoder_for = |_| |Words.(values), writer| {
  Hooks : format
  Hooks.encode_list(writer, values.len(), |items, add| {
   var $items = items
   for value in values {
    $items = add($items, |item| Hooks.encode_str(value, item))?
   }
   Ok($items)
  })
 }
}

log : Port.Command(Words)
log = Port.command({ name: "example.org/words/log" })

sent : List(Engine.Command)
sent = [log.send(Words.(["a", "b"]))]

main : Engine.Program
main = Engine.program(|| Engine.text("public handwritten port codec"))
