app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.PortJson

## A handwritten encoder that writes text the format's hooks did not, and
## resets the nesting count, by building the encoder's output itself.
Forged := [Forged].{
 encoder_for : PortJson.Format -> (Forged, PortJson.Out -> Try(PortJson.Out, []))
 encoder_for = |_| |_, _| Ok(PortJson.Out.{ text: "{{{ not json".to_utf8(), blobs: [], files: [], depth: 0, over: Bool.False })
 parser_for : PortJson.Format -> (PortJson.State -> Try({ value : Forged, rest : PortJson.State }, PortJson.Error))
 parser_for = |_| |state| Ok({ value: Forged.Forged, rest: state })
}

sent : Try(Engine.PortMessage, [TooDeep])
sent = PortJson.encode(Forged.Forged)

main : Engine.Program
main = Engine.program(|| Engine.text("port API contract"))
