app [main] { pf: platform "<PLATFORM>" }

import pf.Engine

Caption := { text : Str }.{
 to_text_signal : Caption -> Engine.Signal(ctx, Str)
 to_text_signal = |{ text }| Engine.const(text)
}

view : () -> Engine.Elem(Engine.Root)
view = || {
 name : Engine.Signal(Engine.Root, Str)
 name = "count"
 count : Engine.Signal(Engine.Root, U64)
 count = Engine.const(7)
 custom : Caption
 custom = { text: "custom" }
 Engine.element("p", [Engine.attr_s("title", "Saved")], [Engine.text_s("${name} ${count.to_text_signal()}"), Engine.text_s("${custom}")])
}

main : Engine.Program
main = Engine.program(view)
