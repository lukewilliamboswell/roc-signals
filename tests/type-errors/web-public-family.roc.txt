app [main] { pf: platform "<PLATFORM>" }
import pf.Engine
import pf.Web
tag : Web.Tag
tag = "my-widget"
attribute : Engine.AttrName
attribute = Web.AttrName.from_str("class") ?? crash "valid computed name"
event : Engine.EventType
event = "vendor:ready"
method : Web.Method
method = "GET"
fetch : Engine.Ask(Engine.Fetched, Engine.FetchFailure)
fetch = Engine.File.fetch({ url: "/", method })
upload : Engine.File -> Engine.Ask(Engine.Response, Engine.HttpFailure)
upload = |file| file.upload({ url: "/" })
main : Engine.Program
main = Engine.program(|| Engine.element(tag, [Engine.attr(attribute, method.text())], [Engine.text(event.text())]))

Store : {}
