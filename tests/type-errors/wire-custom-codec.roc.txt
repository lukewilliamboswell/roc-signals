app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.Wire
import pf.Worker

Words := List(Str).{
 parser_for : _
 ## A handwritten codec names only public state and accumulator types.
 encoder_for : Wire.Format -> (Words, Wire.Enc -> Try(Wire.Enc, []))
 encoder_for = |_| |Words.(values), writer| Wire.Format.encode_list(writer, values.len(), |items, add| write_words(values, items, add))
}

## The container accumulator and each item writer are deliberately distinct.
write_words : List(Str), Wire.Items, (Wire.Items, (Wire.Enc -> Try(Wire.Enc, [])) -> Try(Wire.Items, [])) -> Try(Wire.Items, [])
write_words = |values, initial, add| {
 var $items = initial
 for value in values {
  $items = add($items, |writer| Wire.Format.encode_str(value, writer))?
 }
 Ok($items)
}

job : Worker.Task(Words, Words, Worker.Never)
job = Worker.task({ name: "test/public-wire-codec", run!: |value| Ok(value) })

main : Engine.Program
main = Engine.program(|| Engine.text("public handwritten codec")).with_workers([job.registration()])

## This contract has no shared store.
Store : {}
