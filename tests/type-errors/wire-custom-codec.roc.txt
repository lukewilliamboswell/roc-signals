app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.Worker

Words := List(Str).{
 parser_for : _
 ## A handwritten codec is generic over its format: it names the hooks it
 ## calls, never the platform's format or state types.
 encoder_for : format -> (Words, enc -> Try(enc, []))
  where [
   format.encode_list : enc, U64, (items, (items, (enc -> Try(enc, [])) -> Try(items, [])) -> Try(items, [])) -> Try(enc, []),
   format.encode_str : Str, enc -> Try(enc, []),
  ]
 encoder_for = |_| |Words.(values), writer| {
  Hooks : format
  Hooks.encode_list(writer, values.len(), |items, add| {
   var $items = items
   for value in values {
    $items = add($items, |item| Hooks.encode_str(value, item))?
   }
   Ok($items)
  })
 }
}

job : Worker.Task(Words, Words, Worker.Never)
job = Worker.task({ name: "test/public-wire-codec", run!: |value| Ok(value) })

main : Engine.Program
main = Engine.program(|| Engine.text("public handwritten codec")).with_workers([job.registration()])

## This contract has no shared store.
Store : {}
