app [main] { pf: platform "<PLATFORM>" }

import pf.Engine
import pf.Worker

## A record key's equality is its type's, which no format hook sees, so a
## payload refuses it at `roc check` rather than merging two keys.
job : Worker.Task(Dict({ f : F64 }, Str), U64, Worker.Never)
job = Worker.task({ name: "test/record-key", run!: |value| Ok(value.len()) })

main : Engine.Program
main = Engine.program(|| Engine.text("record dictionary key")).with_workers([job.registration()])
